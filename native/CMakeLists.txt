# Adapted from MLX extension build example (Apple Inc., MIT).
cmake_minimum_required(VERSION 3.27)
project(mtp_ple_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python 3.11 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE nanobind_ROOT COMMAND_ERROR_IS_FATAL ANY)
find_package(nanobind CONFIG REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m mlx --cmake-dir OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE MLX_ROOT COMMAND_ERROR_IS_FATAL ANY)
find_package(MLX CONFIG REQUIRED)
nanobind_add_module(_mtp_ple_native NB_STATIC NB_DOMAIN mlx NOMINSIZE mtp_ple_native.cpp)
target_link_libraries(_mtp_ple_native PRIVATE mlx)
target_link_options(_mtp_ple_native PRIVATE -Wl,-rpath,@loader_path)
